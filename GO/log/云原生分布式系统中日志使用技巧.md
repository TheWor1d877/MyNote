 1. 错误发生处立即记录日志（带上下文） + 返回语义化错误
 2. 上层只做错误传播，不用记录相同错误
 3. info与warn也在业务逻辑处记录，而不是延迟到main

## 分层处理
第一层：业务逻辑层面：
详细记录日志，返回封装之后的错误
```go
func (s *server) Upload(stream pb.FileService_UploadServer) error {
    peerInfo, _ := peer.FromContext(stream.Context())
    log := s.logger.With(zap.String("client", peerInfo.Addr.String()))

    // ... 接收元数据
    if !validFilename(filename) {
        // ✅ warn：客户端问题，记录但不告警
        log.Warn("非法文件名",
            zap.String("filename", filename),
        )
        return status.Error(codes.InvalidArgument, "文件名包含非法字符")
    }

    file, err := os.Create(filepath.Join(storageDir, filename))
    if err != nil {
        // ✅ error：服务端问题，需告警
        log.Error("创建文件失败",
            zap.String("filename", filename),
            zap.Error(err), // 包含完整堆栈（如果用了 zap.Error）
        )
        return status.Error(codes.Internal, "存储服务暂时不可用")
    }

    // ... 写入数据
    if _, err := file.Write(chunk); err != nil {
        log.Error("写入文件失败",
            zap.String("filename", filename),
            zap.Error(err),
        )
        return status.Error(codes.Internal, "磁盘写入失败")
    }

    log.Info("文件上传成功", zap.String("filename", filename))
    return stream.SendAndClose(&pb.UploadResponse{Success: true})
}
```

第二层：gRPC 框架层（自动处理）
gRPC 会自动将 status.Error 转为客户端可见的错误
不需要你在 main 里再处理这些业务错误

第三层：main 函数（只处理致命错误）
```go
func main() {
    logger, _ := zap.NewProduction()
    defer logger.Sync()

    lis, err := net.Listen("tcp", ":50051")
    if err != nil {
        // ✅ 只记录启动阶段的致命错误
        logger.Fatal("监听端口失败", zap.Error(err))
    }

    s := grpc.NewServer()
    pb.RegisterFileServiceServer(s, &server{logger: logger})

    // ✅ 这里的 err 通常是：端口被占、TLS 配置错、系统资源耗尽
    // 不是业务错误！所以需要记录
    if err := s.Serve(lis); err != nil {
        logger.Fatal("gRPC 服务异常退出", zap.Error(err))
    }
}
```
