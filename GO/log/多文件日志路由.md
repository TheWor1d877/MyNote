“多文件日志路由”就是：根据日志的“内容特征”（比如级别、模块、业务类型等），自动把它们写入不同的日志文件中。

## 核心原理: zap的`core`与`LevelEnabler`
Zap 允许你组合多个 zapcore.Core，每个 Core 可以：
- 指定输出目标（文件、Stdout）
- 设置级别过滤器（如只写 ERROR）
- 定义编码格式（JSON / Console）

## 使用案例
```go
package main

import (
	"os"

	"go.uber.org/zap"
	"go.uber.org/zap/zapcore"
	"gopkg.in/natefinch/lumberjack.v2"
)

var (
	MainLogger *zap.Logger
	RaftLogger *zap.Logger
	S3Logger   *zap.Logger
)

func initLogger(logDir string) error {
	if err := os.MkdirAll(logDir, 0755); err != nil {
		return err
	}

	// ====== 1. 主日志：INFO 和 ERROR 分离 ======
	infoWriter := zapcore.AddSync(&lumberjack.Logger{
		Filename:   logDir + "/app.info.log",
		MaxSize:    100, // MB
		MaxBackups: 5,
		MaxAge:     30, // days
		Compress:   true,
	})

	errorWriter := zapcore.AddSync(&lumberjack.Logger{
		Filename: logDir + "/app.error.log", MaxSize: 100,
		MaxBackups: 5,
		MaxAge:     30,
		Compress:   true,
	})

	// 编码器配置
	encoderConfig := zap.NewProductionEncoderConfig()
	encoderConfig.EncodeTime = zapcore.ISO8601TimeEncoder
	encoder := zapcore.NewJSONEncoder(encoderConfig)

	//核心配置
	infoCore := zapcore.NewCore(
		encoder,
		infoWriter,
		zap.LevelEnablerFunc(
			func(lvl zapcore.Level) bool {
				return lvl >= zapcore.InfoLevel && lvl < zapcore.ErrorLevel
			}),
	)

	errCore := zapcore.NewCore(
		encoder,
		errorWriter,
		zap.LevelEnablerFunc(
			func(lvl zapcore.Level) bool {
				return lvl >= zapcore.ErrorLevel
			}))

	// 合并 Core
	mainCore := zapcore.NewTee(infoCore, errCore)
	MainLogger = zap.New(mainCore, zap.AddCaller(), zap.AddStacktrace(zapcore.ErrorLevel))

	RaftLogger = newModuleLogger(logDir+"/raft.log", "raft")
	S3Logger = newModuleLogger(logDir+"/s3.log", "s3")

	return nil
}

func newModuleLogger(filename, module string) *zap.Logger {
	writer := zapcore.AddSync(&lumberjack.Logger{
		Filename:   filename,
		MaxSize:    100,
		MaxBackups: 5,
		MaxAge:     30,
		Compress:   true,
	})

	encoderConfig := zap.NewProductionEncoderConfig()
	encoderConfig.EncodeTime = zapcore.ISO8601TimeEncoder
	encoder := zapcore.NewJSONEncoder(encoderConfig)

	core := zapcore.NewCore(encoder, writer, zapcore.DebugLevel)
	return zap.New(core, zap.AddCaller(), zap.Fields(zap.String("module", module)))
}

// 全局便捷函数
func Info(msg string, fields ...zap.Field) {
	MainLogger.Info(msg, fields...)
}

func Error(msg string, fields ...zap.Field) {
	MainLogger.Error(msg, fields...)
}

```

这样就能在主日志中将不同等级的日志记录到不同的文件里面，也可以创建某个模块专用的logger

```go main.go 
package main

import (
    "your-project/logger" // 替换为你的模块路径
    "go.uber.org/zap"
)

func main() {
    // 初始化日志系统
    if err := logger.InitLogger("./logs"); err != nil {
        panic(err)
    }
    defer logger.MainLogger.Sync()

    // 主日志：自动分离
    logger.Info("服务启动", zap.String("version", "v1.0"))
    logger.Error("数据库连接失败", zap.Error(errExample()))

    // 模块日志：独立文件
    logger.RaftLogger.Info("Leader elected", zap.String("leader", "node-1"))
    logger.S3Logger.Warn("Slow S3 upload", zap.Duration("took", 2*time.Second))
}


func errExample() error {
    return fmt.Errorf("connection timeout")
}
```
